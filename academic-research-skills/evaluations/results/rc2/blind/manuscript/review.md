import csv
from decimal import Decimal as D

with open("figure-results.csv", newline="") as f:
    rows = list(csv.DictReader(f))
by_c = {c: [r for r in rows if r["condition"] == c] for c in ("C1", "C2")}
for r in rows:
    assert D(r["loss_A"]) - D(r["loss_B"]) == D(r["difference_A_minus_B"])
for rs in by_c.values():
    assert sum(D(r["target_weight"]) for r in rs) == 1
    assert sum(int(r["n_clusters"]) for r in rs) == 30
w1 = {r["group"]: D(r["target_weight"]) for r in by_c["C1"]}
def means(rs, weights=None):
    return tuple(sum((weights[r["group"]] if weights else D(r["target_weight"]))
                     * D(r[key]) for r in rs) for key in ("loss_A", "loss_B"))
a1, b1 = means(by_c["C1"])
a2, b2 = means(by_c["C2"])
af, bf = means(by_c["C2"], w1)
assert (a1, b1, a1-b1) == (D("7.40"), D("6.72"), D("0.68"))
assert (a2, b2, a2-b2) == (D("6.74"), D("7.58"), D("-0.84"))
assert (af, bf, af-bf) == (D("7.34"), D("7.46"), D("-0.12"))
assert (a2-b2)-(af-bf) == D("-0.72")
